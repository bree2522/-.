function updateStatus() {

    document.getElementById("status").innerHTML = `
        🌍 환경 점수: ${gameState.eco_power}<br>
        ❤️ 여자친구 호감도: ${gameState.gf_love}<br>
        🐻‍❄️ 북극곰 호감도: ${gameState.bear_love}<br>
        👊 등짝 스매싱 횟수: ${gameState.anger_count}
    `;
}

function renderQuestion() {

    if(gameState.current_stage >= questions.length){
        showEnding();
        return;
    }

    const q = questions[gameState.current_stage];

    document.getElementById("question").innerText = q.question;

    const choicesDiv = document.getElementById("choices");
    choicesDiv.innerHTML = "";

    q.choices.forEach(choice => {

        const btn = document.createElement("button");

        btn.innerText = choice.text;

        btn.onclick = () => {

            choice.effect();

            document.getElementById("message").innerText =
                choice.message;

            gameState.current_stage++;

            updateStatus();

            setTimeout(() => {
                renderQuestion();
            }, 2000);
        };

        choicesDiv.appendChild(btn);

    });

}
